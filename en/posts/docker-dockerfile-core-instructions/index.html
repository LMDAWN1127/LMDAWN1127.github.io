FROM alpine:3.20.2

MAINTAINER dawn 123456@dawn.com www.dawn.com xixi haha

LABEL school=czu \
      class=computer01 \
      auther=dawn \
      email=123456@dawn.com \
      office=www.dawn.com

RUN sed -i 's/dl-cdn.alpinelinux.org/mirrors.tuna.tsinghua.edu.cn/g' /etc/apk/repositories && \
     apk update && \
     apk add nginx && \
     rm -rf /var/cache/ rm -f /usr/share/nginx/*

ADD code/dawn-pingtai.tar.gz /usr/share/nginx/html/pingtai

COPY config/games.conf /etc/nginx/http.d/default.conf

# Specify the working directory; it is auto-switched to when connecting to the container
WORKDIR /usr/share/nginx/html

# Specify the exposed port number
EXPOSE 80

# Create a storage volume for the container's specified directory
VOLUME /usr/share/nginx/html

CMD ["nginx","-g","daemon off;"]

Roles of the three instructions#

InstructionEffectDescription
WORKDIR /usr/share/nginx/htmlSpecify working directorydocker exec auto-cd to this directory after entering the container
EXPOSE 80Declare exposed portWorks with docker run -P to auto-map a random port
VOLUME /usr/share/nginx/htmlDeclare a storage volumeAutomatically creates an anonymous volume; data persists

FROM alpine:3.20.2

MAINTAINER dawn 123456@dawn.com www.dawn.com xixi haha

LABEL school=czu \
      class=computer01 \
      auther=dawn \
      email=123456@dawn.com \
      office=www.dawn.com

WORKDIR /usr/share/nginx/html

EXPOSE 80

VOLUME /usr/share/nginx/html

RUN sed -i 's/dl-cdn.alpinelinux.org/mirrors.tuna.tsinghua.edu.cn/g' /etc/apk/repositories && \
     apk update && \
     apk add nginx openssh-server curl && \
     sed -i 's@#PermitRootLogin prohibit-password@PermitRootLogin yes@' /etc/ssh/sshd_config && \
     ssh-keygen -A && \
     rm -rf /var/cache/ /usr/share/nginx/html/*

# Variables effective at container runtime, valid in both build and run stages; at runtime, use -e to override the value
ENV dawn_USERNAME=root \
    dawn_PASSWORD=321

# Variables effective at image build time, valid only in the build stage; during build you can use the "--build-arg" option to override the value.
ARG dawn_school=czu \
    dawn_class=computer01

RUN touch /dawn-${dawn_USERNAME} && \
    touch /dawn-${dawn_PASSWORD} && \
    touch /dawn-xixi-${dawn_school} && \
    touch /dawn-haha-${dawn_class}

ADD code/dawn-zhiwu.tar.gz /usr/share/nginx/html/zhiwu

COPY config/games.conf /etc/nginx/http.d/default.conf

COPY scripts/start.sh /

# CMD ["nginx","-g","daemon off;"]
CMD ["/bin/sh","-x","/start.sh"]
# CMD ["tail","-f","/etc/hosts"]

root@docker01:/dawn/dockerfile/alpine/zhiwu-ENV-ARG# cat scripts/start.sh
#!/bin/sh

# init root password
if [ -n $dawn_USERNAME ];then {
    USERNAME=${dawn_USERNAME:-root}

    id ${USERNAME}

    if [ $? != 0 ];then {
       adduser ${USERNAME} -D
    }
    fi
}
fi

if [ -n $dawn_PASSWORD ];then {
   PASSWORD=${dawn_PASSWORD:-123}
}
fi

echo $USERNAME:$PASSWORD | chpasswd

# start sshd service
/usr/sbin/sshd

# start nginx service
nginx -g 'daemon off;'

Differences between ENV and ARG#

Comparison itemENVARG
Effective stageBuild + run stageBuild stage only
Runtime overridedocker run -eNot supported
Build-time overrideNot supporteddocker build --build-arg
Enters the imageWritten to image env varsNo (disappears after build)

FROM alpine:3.20.2
...
ADD code/dawn-liferestart.tar.gz /usr/share/nginx/html/liferestart

COPY config/games.conf /etc/nginx/http.d/default.conf

COPY scripts/start.sh /

# Used for the container's health check. When the in-container service runs normally and the container has a health check specified,
# besides the normal state, it also has a health status.
# The status starts as "starting". Each time the health check passes, it becomes "healthy".
# After a certain number of consecutive failures, it becomes "unhealthy".
HEALTHCHECK --interval=1s \
            --timeout=1s \
            --start-period=10s \
            --retries=5 \
            CMD curl -H "host: game01.dawn.com" 127.0.0.1

CMD ["/bin/sh","-x","/start.sh"]

HEALTHCHECK parameters explained#

ParameterDefault valueMeaning
--interval=1s30sHow often to check after the container starts
--timeout=1s30sA single check exceeding this duration is treated as failed
--start-period=10s0sStart delay before checking; failures during this period are not counted as retries
--retries=53How many consecutive failures before being marked unhealthy

4. Differences between ENTRYPOINT and CMD#

Comparison itemCMDENTRYPOINT
Can be overridden by the docker run CLIYes (appending a command overrides)Not overridable by default
Override methodAppend a command directly after docker runMust force with --entrypoint
Used togetherServes as the argument to ENTRYPOINTServes as the main command

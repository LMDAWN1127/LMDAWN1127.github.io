apiVersion: apps/v1
kind: ReplicaSet
metadata:
  name: rs-xiuxian-matchexpressions
spec:
  # Pod의 복제본 수 지정
  replicas: 5
  # Pod 연결 방식 정의
  selector:
    # 라벨 표현식 기반으로 Pod 연결
    matchExpressions:
      # 라벨의 key를 나타냄
    - key: apps
      # 라벨의 value를 나타냄
      values:
      - v1
      - v2
      - v3
      # key와 value 간의 관계 지정: In, NotIn, Exists, DoesNotExist
      #   In:
      #      key 값이 반드시 value 목록 내에 존재해야 함.
      #   NotIn:
      #      In의 반대.
      #   Exists:
      #      key만 존재하면 되며 value는 무관 (value 필드 생략 가능).
      #   DoesNotExist:
      #      지정된 key가 존재하지 않아야 함 (Exists의 반대, value 필드 생략 가능).
      operator: In
  # Pod 템플릿 정의
  template:
    metadata:
      labels:
        apps: v1
        school: czu
        class: computer01
    spec:
       containers:
       - name: c1
         image: harbor:443/dawn/apps:v1

이것이 바로 rs가 rc보다 우수한 점입니다: RS는 5개의 복제본을 선언했지만 새로 생성된 Pod는 2개(cdhjw, q8s66)뿐입니다. matchExpressions의 집합 기반 셀렉터 apps In (v1,v2,v3)가 이미 존재하던 xiuxian01/02/03을 입양(Adopt)했기 때문입니다(기존 Pod 3개 + 신규 Pod 2개 = 5개). 반면 xiuxian04의 라벨은 apps=v4로 In 목록에 없으므로 입양되지 않습니다.

apiVersion: apps/v1
kind: Deployment
metadata:
  name: deloy-xiuxian-update
spec:
  # 업그레이드 전략
  strategy:
    # 업그레이드 유형 지정, "Recreate" 또는 "RollingUpdate"
    #   Recreate:
    #     모든 이전 Pod를 삭제한 후 새 Pod를 생성함. 권장하지 않음.
    type: Recreate
  replicas: 3
  selector:
    matchLabels:
      apps: xiuxian
  template:
    metadata:
      labels:
        apps: xiuxian
        version: v1
    spec:
       containers:
       - name: c1
         image: harbor:443/dawn/apps:v3
         # 이미지 풀 정책 지정, 유효값: Always, Never, IfNotPresent
         #    Never:
         #      로컬에 이미지가 있으면 기동을 시도하고, 없어도 원격에서 당겨오지 않음.
         #    IfNotPresent:
         #      로컬에 이미지가 있으면 기동하고, 없으면 원격 저장소에서 풀함.
         #    Always:
         #      로컬 이미지가 있어도 원격 저장소의 다이제스트와 대조하여 일치하면 로컬 캐시를 사용하고, 불일치하면 새로 풀함.
         #      로컬에 이미지가 없으면 원격 저장소에서 직접 풀함.
         #
         # 기본 정책은 이미지 태그에 따라 결정됨:
         #    태그가 "latest"인 경우 기본값은 "Always";
         #    태그가 "latest"가 아닌 경우 기본값은 "IfNotPresent"
         imagePullPolicy: Always

apiVersion: apps/v1
kind: Deployment
metadata:
  name: deloy-xiuxian-update
spec:
  # 업그레이드 전략
  strategy:
    # 업그레이드 유형 지정, "Recreate" 또는 "RollingUpdate"
    #   Recreate:
    #     모든 이전 Pod를 삭제한 후 새 Pod를 생성함. 권장하지 않음.
    #   RollingUpdate:
    #     롤링 업데이트. Pod 일부를 점진적으로 업데이트하여 이전 Pod를 순차 교체함 (기본값).
    type: RollingUpdate
    # 롤링 업데이트 전략 정의. "type: RollingUpdate"일 때만 유효.
    rollingUpdate:
      # 업그레이드 진행 중 원래 복제본 수보다 초과하여 기동할 수 있는 최대 Pod 수.
      # 백분율과 숫자를 지원하며 미지정 시 기본값은 "25%".
      maxSurge: 2
      # 업그레이드 진행 중 사용할 수 없는 최대 Pod 수.
      # 백분율과 숫자를 지원하며 미지정 시 기본값은 "25%".
      maxUnavailable: 1
  replicas: 5
  selector:
    matchLabels:
      apps: xiuxian
  template:
    metadata:
      labels:
        apps: xiuxian
        version: v1
    spec:

  replicas: 1

        - name: MYSQL_ROOT_PASSWORD

  name: dawn-mysql-svc

2.7 扩展实战：多库备份恢复与双 NFS 卷演练#

  1. 将 WordPress 的数据部署到 K8S 集群外部，WordPress 部署到 K8S 集群内部
  2. 创建测试库：CREATE DATABASE dawn; CREATE TABLE student(id INT PRIMARY KEY AUTO_INCREMENT, name VARCHAR(255) NOT NULL, hobby VARCHAR(255) NOT NULL);
  3. 插入至少 3 条测试数据，包含自己的名字和爱好
  4. 对外部数据库备份，再在 K8S 内部部署 MySQL 并还原
  5. 外部验证 student 表数据是否恢复完整

  template:
    spec:

          server: 11.0.1.231
          path: /dawn/data/nfs-server/scripts/

        image: harbor:443/dawn-db/mysql:8.3.0-oracle
        volumeMounts:
        - name: db
          mountPath: /data
        - name: scripts
          mountPath: /scripts
        command:
        - /bin/bash
        - -x
        - /scripts/backup.sh
      restartPolicy: Never
  backoffLimit: 0

# /dawn/manifests/cj/01-cj-hello.yaml
apiVersion: batch/v1
kind: CronJob
metadata:
  name: hello
spec:
  schedule: "* * * * *"              # 每分钟调度一次
  jobTemplate:
    spec:
      template:
        spec:
          containers:
          - name: hello
            image: harbor:443/dawn-linux/alpine:3.20.2
            imagePullPolicy: IfNotPresent
            command:
            - /bin/sh
            - -c
            - date; echo Hello from the Kubernetes cluster
          restartPolicy: OnFailure

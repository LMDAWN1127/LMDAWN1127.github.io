一、名称空间介绍#

K8S 所谓的名称空间，就是用来隔离 K8S 集群资源的。

[root@master v4-svc-volumes-CoreDNS-ns]# cat 01-rc-mysql-ns-dawn.yaml 
apiVersion: v1
kind: ReplicationController
metadata:
  name: rc-mysql
  namespace: dawn
spec:
  replicas: 1
  selector:
    apps: mysql80
  template:
    metadata:
      labels:
        apps: mysql80
    spec:   
      volumes:
      - name: data
        nfs:
          server: master
          path: /dawn/data/nfs-server/mysql80-dawn
      containers:
      - image: harbor:443/dawn-db/mysql:8.3.0-oracle
        name: db
        volumeMounts:
        - name: data
          mountPath: /var/lib/mysql
        env:
        - name: MYSQL_ROOT_PASSWORD
          value: dawn
        - name: MYSQL_DATABASE
          value: wordpress
        - name: MYSQL_USER
          value: dawn
        - name: MYSQL_PASSWORD
          value: "123"
[root@master v4-svc-volumes-CoreDNS-ns]# cat 02-svc-db-ns-dawn.yaml 
apiVersion: v1
kind: Service
metadata:
  name: db-svc
  namespace: dawn
spec:
  type: ClusterIP
  # 指定ClusterIP的VIP地址，指定svc的网段即可，集群部署采用的svc网段为"10.200.0.0/16"
  # clusterIP: 10.200.0.36
  selector:
    apps: mysql80
  ports:
  - port: 3306
    targetPort: 3306
[root@master v4-svc-volumes-CoreDNS-ns]# cat 03-rc-wordpress.yaml 
apiVersion: v1
kind: ReplicationController
metadata:
  name: rc-wordpress
  namespace: kube-public
spec:
  replicas: 3
  selector:
    apps: wp
  template:
    metadata:
      labels:
        apps: wp
    spec:
      volumes:
      - name: data
        nfs:
          server: master
          path: /dawn/data/nfs-server/wordpress-dawn
      containers:
      - image: harbor:443/dawn-db/wordpress:latest
        name: wp
        volumeMounts:
        - name: data
          mountPath: /var/www/html
        env:
        - name: WORDPRESS_DB_HOST
          # 指定的是Pod的地址当Pod发生变化时，IP地址需要变动
          # value: 10.100.1.60
          # 此处我指定的是ClusterIP的地址，ClusterIP的地址被我固定
          # value: 10.200.0.36
          # 有些情况下svc的地址无法固定或者已经分配，则不建议使用ClusterIP进行通信
          # 可以考虑直接使用svc的名称即可，前提是资源同一名称空间
          # value: db-svc
          # 当数据库和wordpress不在同一个名称空间时，需要写完整的A记录
          # value: db-svc.dawn.svc.dawn.com
          # 也可以简写到名称空间为止即可，svc及后面的配置可以省略
          value: db-svc.dawn
        - name: WORDPRESS_DB_NAME
          value: wordpress
        - name: WORDPRESS_DB_USER
          value: dawn
        - name: WORDPRESS_DB_PASSWORD
          value: "123"

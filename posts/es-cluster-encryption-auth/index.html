output.elasticsearch:
  hosts: 
  - "http://11.0.1.111:9200"
  - "http://11.0.1.112:9200"
  - "http://11.0.1.113:9200"

setup.template.name: "dawn-modules"
setup.template.pattern: "dawn-log*"
setup.template.overwrite: false
